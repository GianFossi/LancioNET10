      SUBROUTINE CONDEN

      REAL LAMDA,NU,MU

            CHARACTER*4 NAME

      COMMON/OPTION/BINAR(50,50),KIJ,IEQ,IM,ISUP,KCRT,KWRT,VL,ICAL

      COMMON/NCODEX/NCOD(50),NAME(50,5),NCST,NCNST,NCIP,NC,NCW

      COMMON/MOLFR/X(50),Y(50),Z(50),ZN(50,2),SOM(2)

      COMMON/CHCURV/NP,IDTT,DLT,IDPP,DLP,TT(50),PP(50)

      COMMON/SYSTEM/IN,IO,KDOS,MDOS,LDOS,NDOS,NUOVER,NOUTP3,KCONV,IDEBUG

      COMMON/INDEX/KBDF,KE,KU,KHS,KW

      COMMON/PARTIZ/LAMDA,MU,NU,YW

      COMMON/COMPRS/ZCR,ZV,ZL,ZLWT

      COMMON/TEMPRE/T0,P0,T,P

      COMMON/BYPASS/KIN

      DATA TCW/647.3/,PCW/222.71/,DT/-0.5/

            IF(IDEBUG.GE.2)WRITE(IO,2000)

      KFLC=0

      ISTP=0

      IWDP=0

      IWBP=0

      NBP=0

      NDP=0

      LDP=0

      LBP=0

      KIN=0

      IDP=0

      KP=0

      IF(KW.NE.0) ZW=Z(NCW)

    1 CONTINUE

      KP=KP+1

      IF(KP.GT.NP) RETURN

      T=TT(KP)

      P=PP(KP)

      T0=T

      P0=P

    5 CONTINUE

      WRITE(IO,1000)

      IF(KP.GT.1) ICAL=1

      IF(LAMDA.GT.0.9) LAMDA=0.9

      IF(LAMDA.LT.0.1) LAMDA=0.1

      IW=0

      IND=7

      CALL PROCEL(IND)

      IF(KCONV) 35,35,10

   10 WRITE(IO,1005)

      WRITE(IO,1006)

      IF(KW.NE.0.AND.MU.GT.0.) IW=1

      CALL PROP(1,2,1,1,IW,0,0)

      CALL OUTPUT(IND,IW,ISTP)

      IF(NDP) 20,20,15

   15 NDP=2

      GO TO 226

   20 IF(NBP) 30,30,25

   25 NBP=2

      GO TO 295

   30 KIN=1

      LDP=0

      LBP=0

      GO TO 1

   35 WRITE(IO,1010)

      WRITE(IO,1006)

      KIN=0

      IF(NBP) 45,45,40

   40 NBP=3

      GO TO 295

   45 IF(LBP) 55,55,50

   50 KBP=0

      IWBP=1

      GO TO 250

   55 IF(NDP) 65,65,60

   60 NDP=3

      GO TO 226

   65 IF(LDP) 75,75,70

   70 LDP=0

      IF((ZV-ZL).GT.0.01.AND.LAMDA.LT.0.0002) GO TO 235

      IF((ZV-ZL).LE.0.01.AND.ZV.LT.ZCR) GO TO 235

      KDP=0

      IWDP=1

      GO TO 200

   75 IF(KW) 110,110,80

   80 CALL TENVAP(PW,DTPW,T,1,0)

      IF(T.GE.TCW.OR.NU.GT.0..OR.(PW/P).GE.1.) GO TO 110

      KFLC=KFLC+1

      IF(KFLC.GT.2) GO TO 305

      WRITE(IO,1090)

      KFL=0

      NU=0.

      IW=0

      KW=4

      IND=6

      CALL PROCEL(IND)

c      IF(KCONV) 300,300,85

      IF(KCONV) 301,301,85

   85 WRITE(IO,1005)

      TFL=T

      IF(MU.GT.0.) IW=1

      CALL PROP(1,2,1,1,IW,0,0)

      CALL OUTPUT(IND,IW,ISTP)

      T=T0

      KW=1

   90 CONTINUE

      IF(TFL-T) 95,1,5

   95 IF(KFL) 96,96,97

   96 WRITE(IO,1000)

      WRITE(IO,1015)

   97 IND=7

      CALL TENVAP(PW,DTPW,T,1,0)

      YW=PW/P

      MU=(ZW-YW)/(1.-YW)

      IF(MU) 300,300,100

  100 IF(KFL) 101,101,102

  101 WRITE(IO,1005)

      WRITE(IO,1020)

  102 LAMDA=1.-MU

      ICAL=1

      NU=0.

      DO 105 I=1,NCW

      Y(I)=Z(I)

  105 X(I)=0.

      Y(NCW)=YW

      CALL PROP(2,2,0,1,1,1,0)

      CALL OUTPUT(IND,1,ISTP)

      KP=KP+1

      IF(KP.GT.1) ICAL=1

      IF(KP.GT.NP) RETURN

      T=TT(KP)

      P=PP(KP)

      T0=T

      P0=P

      KFL=1

      GO TO 90

  110 IF(ISNAN(ZL))GO TO 115

c   aggiunto Lp

      IF(IDEBUG.GE.3)WRITE(IO,2010)ZV,ZL,LAMDA,ZCR

            IF(NDOS.EQ.1)GOTO 235

      IF(ZV.EQ.0..AND.ZL.LE.0..AND.KCONV.EQ.0.AND.LAMDA.GT..9998)GOTO115

c----------------

      IF((ZV-ZL).GE.0.01.AND.LAMDA.GT.0.9998) GO TO 115

      IF((ZV-ZL).LE.0.01.AND.ZL.GT.ZCR) GO TO 115

      GO TO 230

  115 WRITE(IO,1025)

c Se NDOS=1 si è già ricercato il dew point.

      NDOS=1

c Quindi se si ritorna qui bisogna andare al bubble point

      KDP=0

      IND=5

      CALL PROCEL(IND)

      IF(KCONV) 301,301,120

  120 WRITE(IO,1005)

      IF(KW) 150,150,125

  125 IF(T.GE.TCW) GO TO 145

      CALL TENVAP(PW,DTPW,T,1,0)

      IF(ABS(PW-(P*YW)).LE.0.001)GO TO 140

      IF((PW/P)-YW) 130,140,145

  130 IF(.NOT.KCONV.EQ.-1)WRITE(IO,1040)

      PW=P*YW

      CALL TENVAP(PW,DTPW,T,2,0)

      DO 135 I=1,NCW

  135 X(I)=0.

      IW=1

      IDP=3

      GO TO 155

  140 IF(.NOT.KCONV.EQ.-1)WRITE(IO,1030)

      IW=1

      IDP=2

      GO TO 155

  145 IF(.NOT.KCONV.EQ.-1)WRITE(IO,1035)

  150 IDP=1

      IW=0

  155 TDP=T

            IF(KCONV.EQ.-1)THEN

               KCONV=1

               GOTO 176

            ENDIF

      GO TO(160,165,170),IDP

  160 CALL PROP(1,2,1,1,0,0,0)

      GO TO 175

  165 CALL PROP(1,2,1,1,1,0,0)

      GO TO 175

  170 CALL PROP(2,2,0,1,1,1,0)

  175 CALL OUTPUT(IND,IW,ISTP)

  176 IF(IND.EQ.5)T=T0

  180 CONTINUE

      IF(TDP-T) 185,1,200

  185 KDP=1

  190 CONTINUE

      IF(KP.GE.1) ICAL=1

      IND=34

      CALL PROC2(IND)

      KP=KP+1

      IF(NP-KP) 215,195,195

  195 CONTINUE

      T=TT(KP)

      P=PP(KP)

      T0=T

      P0=P

      GO TO 180

  200 IF(KDP) 205,205,210

  205 KDP=-1

      GO TO 190

  210 IF(IDP.EQ.3) GO TO 5

      LDP=1

      GO TO 5

  215 IF(KDP) 220,220,225

  220 IF(IWDP.EQ.0) RETURN

      WRITE(IO,1045)

      WRITE(IO,1050)

      WRITE(IO,1055)

      WRITE(IO,1060)

      WRITE(IO,1065)

      RETURN

  225 NDP=1

      T=TDP+DT

      GO TO 5

  226 IF(NDP.LT.3) RETURN

      WRITE(IO,1045)

      WRITE(IO,1050)

      WRITE(IO,1055)

      WRITE(IO,1065)

      RETURN

  230 CONTINUE

C      IF(IDEBUG.GE.3)WRITE(IO,2010)ZV,ZL,LAMDA,ZCR

      IF((ZV-ZL).GT.0.01.AND.LAMDA.LT.0.0002) GO TO 235

      IF((ZV-ZL).LE.0.01.AND.ZV.LT.ZCR) GO TO 235

      GO TO 305

  235 WRITE(IO,1070)

      KBP=0

      IND=4

      CALL PROCEL(IND)

      IF(KCONV) 300,300,240

  240 WRITE(IO,1005)

      TBP=T

      CALL PROP(1,2,1,1,KW,0,0)

      CALL OUTPUT(IND,KW,ISTP)

      T=T0

  245 CONTINUE

      IF(TBP-T) 265,1,250

  250 IF(KBP) 255,255,260

  255 KBP=-1

      GO TO 270

  260 LBP=1

      GO TO 5

  265 KBP=1

  270 CONTINUE

      IF(KP.GE.1) ICAL=1

      IND=33

      CALL PROC2(IND)

      KP=KP+1

      IF(NP-KP) 280,275,275

  275 CONTINUE

      T=TT(KP)

      P=PP(KP)

      T0=T

      P0=P

      GO TO 245

  280 IF(KBP) 290,290,285

  285 NBP=1

      T=TBP+DT

      GO TO 5

  290 IF(IWBP.EQ.0) RETURN

      WRITE(IO,1045)

      WRITE(IO,1050)

      WRITE(IO,1075)

      WRITE(IO,1080)

      WRITE(IO,1065)

      RETURN

  295 IF(NBP.LT.3) RETURN

      WRITE(IO,1045)

      WRITE(IO,1050)

      WRITE(IO,1075)

      WRITE(IO,1065)

      RETURN

  300 WRITE(IO,1010)

  305 WRITE(IO,1085)

      CALL PROP1(IW)

      ISTP=1

      CALL OUTPUT(IND,IW,ISTP)

      RETURN

  301 WRITE(IO,1011)

            KCONV=-1

            T=10

            IF(KW.GT.0)GOTO 125

            GOTO 150

 1000 FORMAT('1')

 1005 FORMAT(' - CONVERGENCE OBTAINED IN CONDEN')

 1006 FORMAT('   (on flash calculation, T and P given)')

 1010 FORMAT(' - NO CONVERGENCE IN CONDEN')

 1011 FORMAT(' - BEWARE: MAYBE HYDROCARBONS & INERTS SUPERCRITICAL')

 1015 FORMAT(' - BEWARE: FLASH OF SOLE WATER CALCULATION')

 1020 FORMAT(' - BEWARE: LIQUID OF SOLE WATER')

 1025 FORMAT(' - BEWARE: DEW-POINT CALCULATION')

 1030 FORMAT(' - BEWARE: DEW OF HYDROCARBONS AND WATER')

 1035 FORMAT(' - BEWARE: DEW OF SOLE HYDROCARBONS')

 1040 FORMAT(' - BEWARE: DEW OF SOLE WATER')

 1045 FORMAT(1H1,29('*'),' BEWARE ',31('*'))

 1050 FORMAT(' - MAYBE THIS CONDENSATION CURVE CALCULATION HAS ERRORS.')

 1055 FORMAT(' - CHECK WITH A FLASH CALCULATION IF POINTS WITH TEMPERATU

     *RE'/3X,'GREATER THAN CALCULATED DEW-POINT TEMPERATURE CORRESPOND R

     *EALLY'/3X,'A SUPERHEATED GAS, AS SUPPOSED IN THIS CALCULATION.')

 1060 FORMAT(' - CALCULATION FOR POINTS WITH TEMPERATURE LESS THAN'/3X,

     *  'CALCULATED DEW POINT TEMPERATURE ARE SOUND (RETROGRADE GAS).')

 1065 FORMAT(' ',68('*'))

 1070 FORMAT(' - BEWARE: BUBBLE-POINT CALCULATION')

 1075 FORMAT(' - CHECK WITH A FLASH CALCULATION IF POINTS WITH TEMPERATU

     *RE'/3X,'GREATER THAN CALCULATED BUBBLE-POINT TEMPERATURE CORRESPON

     *D REALLY'/3X,'A RETROGRADE LIQUID, AS SUPPOSED IN THIS CALCULATION

     *.')

 1080 FORMAT(' - CALCULATION FOR POINTS WITH TEMPERATURE LESS THAN'/3X,

     *  'CALCULATED BUBBLE-POINT TEMPERATURE ARE SOUND (SUBCOOLED LIQUID

     *).')

 1085 FORMAT(' - PROBLEM NOT SOLVED')

 1090 FORMAT(' - BEWARE: FLASH CALCULATION (TYPE 6.C)')

 2000 FORMAT(' 222 SUBROUTINE CONDEN 222 Entrata')

 2010 FORMAT('     333 SUBROUTINE CONDEN 333'/

     1' ZV=',E12.5,',ZL=',E12.5,',LAMDA=',E12.5,',ZCR=',E12.5)

      END
