Subroutine SOLUZ

Autore

 

Revisioni

 

Scopo

Risolvere l’equazione cubica nell’incognita z (fattore di comprimibilità) che rappresenta l’equazione di stato del sistema liquido-vapore.

Argomenti in uscita

ZSOL

Valore della soluzione scelta

DELT1

Se positivo, l’equazione ha tre soluzioni reali; se negativo, l’equazione ha una sola soluzione.

Argomenti in ingresso

A1

Coefficiente A

B1

Coefficiente B

IFASE

L’equazione di stato è applicata alla fase liquida (IFASE=1) o alla fase aeriforme (IFASE=2).

Listato

C     - SUBROUTINE SOLUZ - 68

C ----------------------------------------------------------------------

      SUBROUTINE SOLUZ(ZSOL,DELT1,A1,B1,IFASE)

      REAL ZSOL,DELT1,A1,B1

      INTEGER*2 IFASE

c      COMMON/COMPRS/ZCR,ZV,ZL,ZLWT

      INCLUDE 'GENERAL.FI'

C      COMMON/SYSTEM/IN,IO,KDOS,MDOS,LDOS,NDOS,NUOVER,NOUTP3,KCONV,IDEBUG

C-------------------------------------------------------------------

      REAL*8 Aloc,Bloc,ARGM1,ARGM2,Ploc,Qloc,DELTAloc,DELT2,ABloc

      REAL*8 SUM1,SUM2,DEN1,DEN2,UNTERZO,ZV,ZL,Z3

C-------------------------------------------------------------------

      Aloc=A1

      Bloc=B1

      UNTERZO=DBLE(1.)/3.

      IF(IDEBUG.GE.6)WRITE(IO,2000)IEQ,A1,B1,IFASE

      GOTO (1,6),IEQ

C    1 P=0.11111111111111-(A-B-B**2)/3.

C      Q=0.03703703703703-(A-B-B**2)/6.+A*B/2.

    1 Ploc=UNTERZO/3.-(Aloc-Bloc-Bloc**2)/3.

      Qloc=UNTERZO/9.-(Aloc-Bloc-Bloc**2)/6.+Aloc*Bloc/2.

      DELTAloc=Qloc*Qloc-Ploc*Ploc*Ploc

      DELT1=DELTAloc

      ABloc=Aloc-Bloc-Bloc*Bloc

      DELT2=(-27*ABloc*ABloc+108*ABloc*ABloc*ABloc+108*Aloc*Bloc

      1+729*Aloc*Aloc*Bloc*Bloc-486*Aloc*Bloc*ABloc)/2916.

C           WRITE(IO,'(''P,Q,DELTAloc******''4E12.5)')P,Q,DELTAloc,DELT2

      IF (DELTAloc)4,3,2

    2 SUM1=Qloc+DSQRT(DELTAloc)

      SUM2=Qloc-DSQRT(DELTAloc)

      DEN1=ABS(SUM1)

      DEN2=ABS(SUM2)

      ZSOL=UNTERZO+(SUM1/DEN1)*(DEN1**UNTERZO)

     *+(SUM2/DEN2)*(DEN2**UNTERZO)

C      WRITE(IO,'('' ZSOL=''E12.5)')ZSOL

      GO TO 12

    3 ZV=UNTERZO+2.*DSQRT(Ploc)

      ZL=UNTERZO-DSQRT(Ploc)

C      WRITE(IO,'('' ZVZL=''2E12.5)')ZV,ZL

      GO TO 10

c    4 ARGM1=0.333333333333333*DACOS(Q/(P*DSQRT(P)))

    4 ARGM1=DACOS(Qloc/(Ploc*DSQRT(Ploc)))/3.

      ARGM2=3.14159265369

      ZV=UNTERZO+DSQRT(Ploc)*DCOS(ARGM1)*2.

C      ZL=1./3.+2*DSQRT(P)*DCOS(ARGM1+0.66666666666666*ARGM2)

      ZL=UNTERZO+DSQRT(Ploc)*DCOS(ARGM1+ARGM2*2./3.)*2.

      Z3=UNTERZO+DSQRT(Ploc)*DCOS(ARGM1+ARGM2/3.)*2.

C     AGGIUNTO LP

C      IF(ZL.LT.0)ZL=0

C---------------------

C      WRITE(IO,'('' ZV1ZL1=''3E12.5)')ZV,ZL,Z3

      GO TO 10

    6 Ploc=(1.-Bloc)**2/9.-(Aloc-3*Bloc**2-2*Bloc)/3.

      Qloc=(1.-Bloc)**3/27.-(1.-Bloc)*(Aloc-3*Bloc**2-2*Bloc)/6.

     1+Aloc*Bloc-Bloc**2-Bloc**3)/2.

      DELTAloc=Qloc*Qloc-Ploc*Ploc*Ploc

      DELT1=DELTAloc

      IF(DELTAloc) 9,8,7

    7 SUM1=Qloc+DSQRT(DELTAloc)

      SUM2=Qloc-DSQRT(DELTAloc)

      DEN1=ABS(SUM1)

      DEN2=ABS(SUM2)

      ZSOL=(1.-Bloc)/3.+(SUM1/DEN1)*(DEN1**UNTERZO)

     *+(SUM2/DEN2)*(DEN2**UNTERZO)

      GO TO 12

    8 ZV=(1.-Bloc)/3.+2*DSQRT(Ploc)

      ZL=(1.-Bloc)/3.-DSQRT(Ploc)

      GO TO 10

    9 ARGM1=UNTERZO*DACOS(Qloc/(Ploc*DSQRT(Ploc)))

      ARGM2=3.14159265369

      ZV=(1.-Bloc)/3.+DSQRT(Ploc)*DCOS(ARGM1)*2.

      ZL=(1.-Bloc)/3.+DSQRT(Ploc)*DCOS(ARGM1+UNTERZO*ARGM2)*2.

   10 IF(IFASE.EQ.1) ZSOL=ZL

      IF(IFASE.EQ.2) ZSOL=ZV

   12 CONTINUE

      IF(IDEBUG.GE.6)WRITE(IO,2010)DELTAloc,ZSOL,ZV,ZL

      RETURN

 2000 FORMAT(T30,'666 SOLUZ 666 - Entrata. IEQ=',I3,',A1=',E12.5,

     *',B1=',E12.5,',IFASE=',I3)

 2010 FORMAT(T30,'666 SOLUZ 666 - Uscita. DELTA=',F9.5,',ZSOL=',F9.5,

     *',ZV=',F9.5,',ZL=',F9.5)

      END
