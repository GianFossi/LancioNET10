2)

·        2:        il criterio ZV¹ZL non è soddisfatto

C ----------------------------------------------------------------------

C     - SUBROUTINE CRICOV - 77

C ----------------------------------------------------------------------

      SUBROUTINE CRICOV(ITEST)

            CHARACTER*4 NAME

      COMMON/OPTION/BINAR(50,50),KIJ,IEQ,IM,ISUP,KCRT,KWRT,VL,IOPT1

      COMMON/NCODEX/NCOD(50),NAME(50,5),NCST,NCNST,NCIP,NC,NCW

      COMMON/FER/F(50),FI(50),SNOR,GI,ERRE,SGI,SGR

      COMMON/INDEX/KBDF,KE,KU,KHS,KW

      COMMON/COMPRS/ZCR,ZV,ZL,ZLW

      COMMON/SYSTEM/IN,IO,KDOS,MDOS,LDOS,NDOS,NUOVER,NOUTP3,KCONV,IDEBUG

      ITEST=0

      IF(KE.EQ.5) GO TO 15

      IF(KBDF.LT.3) GO TO 5

      DO 1 I=1,NC

      IF(ABS(F(I)).GE.VL)THEN

             IF(IDEBUG.GE.5)WRITE(IO,2000)I,F(I),VL

             RETURN

      ENDIF

    1 CONTINUE

    5 DO 10 I=1,NC

      IF(ABS(FI(I)).GE.VL)THEN

             IF(IDEBUG.GE.5)WRITE(IO,2010)I,FI(I),VL

             RETURN

      ENDIF

   10 CONTINUE

      IF(ABS(SNOR).GE.VL)THEN

             IF(IDEBUG.GE.5)WRITE(IO,2020)SNOR,VL

             RETURN

      ENDIF

   15 IF(KHS.EQ.0) GO TO 20

      IF(ABS(GI).GE.(SGI*VL))THEN

             IF(IDEBUG.GE.5)WRITE(IO,2030)GI,SGI,VL

             RETURN

      ENDIF

   20 IF(KE.GT.2) GO TO 25

      IF(ABS(ERRE).GE.(SGR*VL))THEN

             IF(IDEBUG.GE.5)WRITE(IO,2040)ERRE,SGR,VL

             RETURN

      ENDIF

   25 GO TO(30,30,30,30,35),KE

   30 IF(ABS(ZV-ZL).LE..001) GO TO 55

      GO TO 50

   35 GO TO(40,45),KBDF

   40 CONTINUE

      GO TO 50

   45 CONTINUE

   50 ITEST=1

      RETURN

   55 ITEST=2

      IF(IDEBUG.GE.5)WRITE(IO,2050)ZV,ZL

      RETURN

 2000 FORMAT(' 555 CRICOV 555 - F(I)>VL'/1X,I5,2E12.5)

 2010 FORMAT(' 555 CRICOV 555 - FI(I)>VL'/1X,I5,2E12.5)

 2020 FORMAT(' 555 CRICOV 555 - SNOR>VL'/1X,2E12.5)

 2030 FORMAT(' 555 CRICOV 555 - GI>SGI*VL'/1X,3E12.5)

 2040 FORMAT(' 555 CRICOV 555 - ERRE>SGR*VL'/1X,3E12.5)

 2050 FORMAT(' 555 CRICOV 555 - ZV=ZL'/1X,2E12.5)

      END
