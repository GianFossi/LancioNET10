Subroutine PROC3

Autore

 

Revisioni

 

Scopo

Gestisce la soluzione di problemi con IND=35 e 36, cioè la costruzione delle curve di riscaldamento e di raffreddamento.

Argomenti

IND

Indicatore del tipo di calcolo. Vedi variabile IND

Listato

C ----------------------------------------------------------------------

C     - SUBROUTINE PROC3 - 16

C ----------------------------------------------------------------------

      SUBROUTINE PROC3(IND)

      COMMON/TITLE/TIT(18),IUF,IUM,IUP,INDI,FTM,WTF,HFT,QT,SFT,AFHS(3,4)

      COMMON/CHCURV/NP,IDT,DT,IDP,DP,TT(50),PP(50)

      COMMON/SYSTEM/IN,IO,KDOS,MDOS,LDOS,NDOS,NUOVER,NOUTP3,KCONV,IDEBUG

C*****MODIFICA FBM-HI

      CHARACTER*4 A1,A2,BLK,TIT

C*****FINE MODIFICA

      DIMENSION A1(2),A2(3)

      DATA A1/'HEAT','COOL'/,A2/'ING ','CURV','E   '/,IK/15/,BLK/'    '/

      IF(IDEBUG.GE.1)WRITE(IO,1100)IND

      DO 55 I=15,18

C*****MODIFICA FBM-HI

CM    IF(TIT(I)-BLK) 60,55,60

      IF(TIT(I).NE.BLK) GOTO 60

   55 CONTINUE

      KK=IND-34

      TIT(IK)=A1(KK)

      DO 50 JK=1,3

C*****MODIFICA FBM-HI

      ICONT=IK+JK

CM 50 TIT(IK+JK)=A2(JK)

   50 TIT(ICONT)=A2(JK)

   60 I=0

      IF(IND-35) 1,1,25

    1 IF(IDT) 15,15,5

    5 I=I+1

      IF(NP-I) 15,15,10

   10 IF(TT(I).LT.TT(I+1)) GO TO 5

      GO TO 20

   15 CALL EVAPOR

      RETURN

   20 WRITE(IO,100)

      WRITE(IO,105)

      RETURN

   25 IF(IDT) 40,40,30

   30 I=I+1

      IF(NP-I) 40,40,35

   35 IF(TT(I).GT.TT(I+1)) GO TO 30

      GO TO 45

   40 CALL CONDEN

      RETURN

   45 WRITE(IO,100)

      WRITE(IO,110)

      RETURN

  100 FORMAT(' ****** BEWARE: ERROR IN INPUT CARDS',/)

  105 FORMAT(' - THE TEMPERATURES ASSIGNED ARE NOT INCREASING')

  110 FORMAT(' - THE TEMPERATURES ASSIGNED ARE NOT DECREASING')

 1100 FORMAT(' *** SUBROUTINE PROC3 *** Entrata'/

     1'     IND=',I5)

      END
