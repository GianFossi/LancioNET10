      SUBROUTINE PROCIT(IND)

      COMMON/SYSTEM/IN,IO,KDOS,MDOS,LDOS,NDOS,NUOVER,NOUTP3,KCONV,IDEBUG

      COMMON/OPTION/BINAR(50,50),KIJ,IEQ,IM,ISUP,KCRT,KWRT,VL,IOPT1

      COMMON/ACRISI/ICR,ICC,ICK,KSTOP,ISTOP2,IP1,IP2

      COMMON/K00K/ITER,K004,K005,K006,AKK(6)

      COMMON/INDEX/KBDF,KE,KU,KHS,KW

      COMMON/TEMPRE/TINP,PINP,T,P

            LOGICAL*2 RISULT

      KCONV=0

      ITMAX=40

      IF(IDEBUG.GE.3)WRITE(IO,2011)IND,ITER

      IF(ITER.NE.0) ITMAX=ITER

      DO 20 IT=1,ITMAX

      IF(IDEBUG.GE.4)WRITE(IO,2010)T,ICR,IT

      IF(ICR.NE.0)GO TO 1

      IF(IT.GT.1)GO TO 5

    1 CALL THERMO(IEQ,IM,ISUP,1)

      IF(ICR.NE.0) RETURN

      CALL FUNZER

    5 CALL FATNOR

      IF(IT-1) 10,10,15

   10 CALL FUNZOB(FOB)

      CALL PRINT(KWRT,0,0.,0.,FOB)

   15 CALL STANDA(RISULT)

      IF(.NOT.RISULT)THEN

               KCONV=0

               RETURN

            ENDIF

      CALL PROCOV(KTEST,TETA,TETAMN,FOB,IND)

      IF(NUOVER.GT.0)RETURN

      CALL PRINT(KWRT,IT,TETA,TETAMN,FOB)

      IF(KTEST.EQ.1)THEN

         IF(IDEBUG.GE.4)WRITE(IO,2012)

               RETURN

      ENDIF

      CALL CRICOV(ITEST)

      IF(ITEST) 20,20,25

   20 CONTINUE

      RETURN

   25 GO TO(30,35),ITEST

   30 KCONV=1

   35 RETURN

 2010 FORMAT('         444 SUBROUTINE PROCIT 444. T=',F12.3,

     1'(ICR=',I2,',iterazione=',I2,')')

 2011 FORMAT(' 333 SUBROUTINE PROCIT 333. Entrata.'/

     1      'IND=',I3,',ITER=',I3)

 2012 FORMAT('         444 SUBROUTINE PROCIT 444. KTEST da PROCOV = 1')

      END
