Subroutine PROC2

Autore

 

Revisioni

 

Scopo

Gestisce la soluzione di problemi con IND=33 e IND=34, cioè la ricerca delle proprietà dei fluidi.

Argomenti

IND

Indicatore del tipo di calcolo. Vedi variabile IND

Listato

C ----------------------------------------------------------------------

C     - SUBROUTINE PROC2 - 55

C ----------------------------------------------------------------------

      SUBROUTINE PROC2(IND)

      REAL LAMDA,MU,NU,K

      CHARACTER*4 NAME

      COMMON/NCODEX/NCOD(50),NAME(50,5),NCST,NCNST,NCIP,NC,NCW

      COMMON/KVALUE/DTK(50),DPK(50),DXK(50,50),DYK(50,50),K(50)

      COMMON/MOLFR/X(50),Y(50),Z(50),ZN(50,2),SOM(2)

      COMMON/INDEX/KBDF,KE,KU,KHS,KW

      COMMON/PARTIZ/LAMDA,MU,NU,YW

      COMMON/SYSTEM/IN,IO,KDOS,MDOS,LDOS,NDOS,NUOVER,NOUTP3,KCONV,IDEBUG

      COMMON/DENSLV/ROL,ROV,TPCL,TPCV,PPCL,PPCV,IROL

      COMMON/TEMPRE/TINP,PINP,T,P

      IF(IDEBUG.EQ.1)WRITE(IO,1100)IND

      MU=0.

      IW=0

      ISTP=0

      DO 1 I=1,NCW

      X(I)=0.

      K(I)=0.

    1 Y(I)=0.

      KHS=3

      KE=5

      KU=4

   21 KBDF=IABS(IND)-32                        !22/06/01

      GO TO (5,20), KBDF

    5 LAMDA=0.

      DO 10 I=1,NC

   10 X(I)=Z(I)

      IF(KW.EQ.0) GO TO 15

      MU=Z(NCW)

      IW=1

   15 NU=1.-LAMDA-MU

      CALL PROP(1,1,1,0,KW,1,1)

      GO TO 30

   20 DO 25 I=1,NCW

   25 Y(I)=Z(I)

      LAMDA=1.

      NU=1.-LAMDA-MU

      CALL PROP(2,2,0,1,0,1,0)

   30 CONTINUE

      IF(IND.LT.0)THEN                        !22/06/01

               IF(KBDF.EQ.1.AND.T.GT.TPCL)IND=34    !22/06/01

               IF(KBDF.EQ.2.AND.T.LT.TPCV)IND=33    !22/06/01

         IF(IND.GT.0)GOTO 21                  !22/06/01

      ENDIF                                   !22/06/01

      CALL OUTPUT(IND,IW,ISTP)

      IF(IDEBUG.EQ.1)WRITE(IO,1110)IND

      RETURN

 1100 FORMAT(' *** SUBROUTINE PROC2 *** Entrata. IND=',I5)

 1110 FORMAT(' *** SUBROUTINE PROC2 *** Uscita.  IND=',I5)

      END
