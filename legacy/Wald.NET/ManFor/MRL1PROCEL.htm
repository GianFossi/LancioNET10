Subroutine PROCEL

Autore

 

Revisioni

 

Scopo

Gestisce la soluzione di problemi con IND<33, cioè tutti i problemi ad eccezione delle curve di riscaldamento e di raffreddamento e ad eccezione della ricerca delle proprietà dei fluidi.

Argomenti

IND

Indicatore del tipo di calcolo. Vedi variabile globale IND.

Listato

C ----------------------------------------------------------------------

C     - SUBROUTINE PROCEL - 83

C ----------------------------------------------------------------------

      SUBROUTINE PROCEL(IND)

      COMMON/SYSTEM/IN,IO,KDOS,MDOS,LDOS,NDOS,NUOVER,NOUTP3,KCONV,IDEBUG

      COMMON/ACRISI/ICR,ICC,ICK,KSTOP,ISTOP2,IP1,IP2

      COMMON/INDEX/KBDF,KE,KU,KHS,KW

      COMMON/BYPASS/KIN

      COMMON/COMPRS/ZCR,ZV,ZL,ZLWT

      COMMON/TEMPRE/TINP,PINP,T,P

      DIMENSION INDICE(32,4)

      DATA INDICE/1,2,3,1,2,2*3,1,2,3*3,1,2,3*3,1,2,5*3,1,2,1,2,1,2,1,2,

     *7*4,10*3,5*2,2*1,8*5,3*1,3*2,4,2*3,1,2,3*3,1,2,3*3,1,2,3*3,2*1,2*2

     *,2*1,2*2,7*0,5*1,5*2,5*0,1,2,4*1,4*2/

      ISTOP2=0

      KSTOP=0

      IP1=0

      IP2=0

      ICK=0

      ICR=0

      ICC=1

      KBDF=INDICE(IND,1)

      KE=INDICE(IND,2)

      KU=INDICE(IND,3)

      KHS=INDICE(IND,4)

            IF(IDEBUG.GE.3)WRITE(IO,2000)IND,KBDF,KE,KU,KHS,KIN

      IF(KIN) 1,1,3

    1 CALL PROCIN(ICR)

    3 CALL PROCIT(IND)

      IF(IDEBUG.GE.4)WRITE(IO,2010)T,ICR,IND

      IF(NUOVER.GT.0)RETURN

      IF(ICR) 5,5,1

    5 RETURN

 2000 FORMAT(' 333 SUBROUTINE PROCEL 333 Entrata'/

     1T10,'IND=',I3,T20,'KBDF=',I3,T30,'KE=',I3,T40,'KU=',I3,T50,'KHS=',

     2I3,T60,'KIN='I3)

 2010 FORMAT('         444 SUBROUTINE PROCEL 444. T=',F12.3,

     1'(ICR=',I2,',IND=',I2,')')

      END
