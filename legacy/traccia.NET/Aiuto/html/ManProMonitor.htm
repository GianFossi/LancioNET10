L'oggetto privato Monitor

L’oggetto privato Monitor ha lo scopo di gestire gli eventi generati dall’oggetto ‘Motore’, che è esposto dalla dll delle routines di base del sistema Lancio, di nome RoutBase1.dll.

Listato:

Option Explicit

Public Routines As RoutBase1.Routines

Public InterrompiTrasferimento As Boolean

Public WithEvents Motore As RoutBase1.Motore

Private Sub Motore_ProgrFine(f As Boolean)

    If Not f Then InterrompiTrasferimento = True

    iAction = 0

End Sub

Private Sub Motore_Cambiofile(f As String)

           Dim i As Integer, icome As String

           InitDaTos

           icome = f

           icome = Left$(icome, Len(icome) - 4)

           Tracciatura.Modo = 1

           Loadd icome 'Carica icome

           frmTraccia.Caption = clsAbout.ProgName + " " + icome

           frmTraccia.AggSecondaPagina

End Sub

Private Sub Motore_NuovoLav()

           SottoCarica Motore.Inizio.Archdir + "\TRACSTD"

End Sub

Private Sub Motore_Rifiuto()

           Dim i As Integer

           frmTraccia.Pagina(0).Value = False

End Sub

Private Sub Motore_SalvaData(f As String)

           DaTos(0).commes = Left$(f, Len(f) - 4)

           SaveInp

End Sub
