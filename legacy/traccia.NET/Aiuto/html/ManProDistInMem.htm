Subroutine DistInMem

Autore:                        Leonardo Presciuttini

Data:                        28/05/2000

Scopo:

Crea una lista ordinata dei fori (caratterizzati dalle coordinate (PuntiD),  dal numero di fila (Fila), dal numero di settore (Settore), dalla distanza dal centro (DistCentro) e dalla posizione nel record buc (Posizbu)). Il criterio di ordinamento è in prima istanza la distanza dal centro decrescente. In seconda istanza, per tutti i fori aventi pari distanza dal centro (tolleranza di uguaglianza 0,1 mm) il criterio di ordinamento prevede che, ove possibile, due fori, uno successivo all’altro, non devono appartenere allo stesso quadrante.  

Chiamata da Bilancio.

Listato:

Sub DistInMem()

       Dim iDat As Integer, iTubo As Integer, ntubi As Integer

       Dim k As Integer, j As Integer, nnn As Integer

       Dim x As Single, y As Single, iCod As Integer

       Dim i As Integer, ii As Integer, iS1 As Integer, iS2 As Integer

       For iDat = 0 To 1

       Tracciatura.LungPIP = iDat + 1

       apri5

       iTubo = 0

       For k = 1 To DaTos(iDat).nset

           For j = DaTos(iDat).inrow(k) To DaTos(iDat).iFrow(k)

              If j > 0 Then

                 Get #Tracciatura.File5, j, DaTos(iDat).bu

                 For nnn = 1 To DaTos(iDat).bu.ici

                    Call CoorXY(iDat, j, nnn, x, y, iCod)

                    If iCod = 0 Then

                       iTubo = iTubo + 1

                       PuntiD(iTubo, iDat).x = x

                       PuntiD(iTubo, iDat).y = y

                       DistCentro(iTubo, iDat) = Sqr(x * x + y * y)

                       Fila(iTubo, iDat) = j

                       Posizbu(iTubo, iDat) = nnn

                       Settore(iTubo, iDat) = k

                    End If

                 Next nnn

              End If

           Next j

       Next k

       ntubi = iTubo

       For iTubo = 1 To ntubi - 1

          Monitor.Motore.Avanzamento = (iTubo + ntubi * iDat) / 2 / ntubi * 0.5 * 100

          For i = iTubo + 1 To ntubi

              If iTubo <> i Then

                  If iDat = 0 Then

                     If DistCentro(iTubo, iDat) < DistCentro(i, iDat) Then

                       ii = i: GoSub Scambia

                     End If

                  Else

                     If DistCentro(iTubo, iDat) < DistCentro(i, iDat) Then

                       ii = i: GoSub Scambia

                     End If

                  End If

              End If

          Next

       Next

       For iTubo = 1 To ntubi - 1

          Monitor.Motore.Avanzamento = (0.5 + (iTubo + ntubi * iDat) / 2 / ntubi) * 0.3 * 100

          For i = iTubo + 1 To ntubi

              If Abs(DistCentro(i, iDat) - DistCentro(iTubo, iDat)) < 0.1 Then

              Else

                    If i > iTubo + 3 Then

                       For j = iTubo To i - 1

                            iS1 = iSector(PuntiD(j, iDat))

                            For k = j + 1 To i

                            iS2 = iSector(PuntiD(k, iDat))

                            If iOkk(iS1, iS2) Then Exit For

                            Next

                            If k > j + 1 Then

                               iTubo = k: ii = j + 1: GoSub Scambia

                            End If

                       Next

                    End If

                 iTubo = i

                 Exit For

              End If

          Next

       Next

       Next iDat

       Tracciatura.LungPIP = 1

       apri5

Exit Sub

Scambia:

                       SWAP PuntiD(iTubo, iDat).x, PuntiD(ii, iDat).x

                       SWAP PuntiD(iTubo, iDat).y, PuntiD(ii, iDat).y

                       SWAP DistCentro(iTubo, iDat), DistCentro(ii, iDat)

                       SWAP Fila(iTubo, iDat), Fila(ii, iDat)

                       SWAP Posizbu(iTubo, iDat), Posizbu(ii, iDat)

                       SWAP Settore(iTubo, iDat), Settore(ii, iDat)

Return

End Sub
