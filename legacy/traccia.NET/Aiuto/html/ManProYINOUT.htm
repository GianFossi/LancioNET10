Subroutine YINOUT

Autore:                       

Data:                        20/05/2000

Scopo:

Calcola l’altezza delle zone libere in ingresso (DaTos(0).yin) e in uscita (DaTos(0).yout) in funzione dei diametri, densità e portate. Per la teoria vedi il manuale teorico.

Listato:

Sub YINOUT()

        Dim ACCAI As Single, ATRIN As Single

        Dim ACCAO As Single, ATROUT As Single

        Dim ROV2IN As Single, ROV2OU As Single, CHECK As Single, Check1 As Single

        Dim ACYLIN As Single, ACYLOU As Single

        If DaTos(0).di1 > DaTos(0).dtin Then DaTos(0).Epsilo = DaTos(0).di1 / 4 - Sqr(DaTos(0).di1 ^ 2 - DaTos(0).dtin ^ 2) / 4

        If DaTos(0).di1 > DaTos(0).dtout Then DaTos(0).Epsil1 = DaTos(0).di1 / 4 - Sqr(DaTos(0).di1 ^ 2 - DaTos(0).dtout ^ 2) / 4

        If DaTos(0).yin <> 0 Then GoTo YI7

        If DaTos(0).dtin <> 0 Then GoTo YI4

        DaTos(0).yin = 0

        GoTo YI7

YI4:    If DaTos(0).roin > 0 And DaTos(0).win > 0 Then GoTo YI5

        ACCAI = DaTos(0).dtin / 4

        DaTos(0).yin = Int(2 * (DaTos(0).Epsilo + ACCAI + 9 - DaTos(0).Diadif / 2)) / 2

        GoTo YI7

YI5:    ATRIN = DaTos(0).dtin ^ 2 * PI / 4 / 1000000

        ROV2IN = DaTos(0).roin * 0.672 * (DaTos(0).win / (DaTos(0).roin * ATRIN * 3600)) ^ 2

        CHECK = 3000

        If ROV2IN < CHECK Then CHECK = ROV2IN

        ACYLIN = DaTos(0).win / (Sqr(CHECK / (DaTos(0).roin * 0.672)) * DaTos(0).roin * 3600) * 1000000

        ACCAI = ACYLIN / (DaTos(0).dtin * PI)

        DaTos(0).yin = Int(2 * (DaTos(0).Epsilo + ACCAI + 9 - DaTos(0).Diadif / 2)) / 2

YI7:    If DaTos(0).yout <> 0 Then GoTo YI10

        If DaTos(0).dtout <> 0 Then GoTo YI8

        DaTos(0).yout = 0

        GoTo YI10

YI8:    If DaTos(0).roout > 0 And DaTos(0).wout > 0 Then GoTo YI9

        ACCAO = DaTos(0).dtout / 6

        DaTos(0).yout = Int(2 * (DaTos(0).Epsil1 + ACCAO - DaTos(0).Diadif / 2)) / 2

        GoTo YI10

YI9:    ATROUT = DaTos(0).dtout ^ 2 * PI / 4 / 1000000

        ROV2OU = DaTos(0).roout * 0.672 * (DaTos(0).wout / (DaTos(0).roout * ATROUT * 3600)) ^ 2

        Check1 = 3000

        If ROV2OU < Check1 Then Check1 = ROV2OU

        ACYLOU = DaTos(0).wout / (Sqr(Check1 / (DaTos(0).roout * 0.672)) * DaTos(0).roout * 3600) * 1000000

        ACCAO = ACYLOU / (DaTos(0).dtout * PI)

        DaTos(0).yout = Int(2 * (DaTos(0).Epsil1 + ACCAO - DaTos(0).Diadif / 2)) / 2

YI10:   If DaTos(0).yin < 0 Then DaTos(0).yin = 0

        If DaTos(0).yout < 0 Then DaTos(0).yout = 0

End Sub
