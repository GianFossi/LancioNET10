Subroutine DisTaglio

Autore:            

Data:                 22/04/2000

Scopo:

Disegna la rappresentazione in sezione del taglio diaframmi.

Chiamata da:          Sub DisTagl, Sub mnuToglti, Sub MuoviTaglio.

Listato:

Public Sub DisTaglio(j As Integer)

        Dim Arg As Single, PXX As Single

        Arg = (DaTos(0).Didia / 2) * (DaTos(0).Didia / 2) - DaTos(0).tagli(j) * DaTos(0).tagli(j)

        If Arg > 0 Then PXX = Sqr(Arg) Else PXX = 0

        Select Case DaTos(0).dt(31)    'angolo taglio

        Case 0

           Monitor.Routines.tratto -PXX, DaTos(0).tagli(j), PXX, DaTos(0).tagli(j), 0, 3

        Case 90

           Monitor.Routines.tratto DaTos(0).tagli(j), PXX, DaTos(0).tagli(j), -PXX, 0, 3

        End Select

End Sub
