Subroutine Sealing

Autore:            

Data:                 22/04/2000

Scopo:

Prepara i dati per la disegnazione delle sealing strips.

Chiamata da:          Sub SubTraccia.

Listato:

Sub SEALING()

    Dim isk As Integer, SEA2 As String

    Dim PX(1 To 4) As Single

    Dim PY(1 To 4) As Single

'7=alfa  10=-1 alfa -2 0 -3 90 -4 180 -5 270 se non definito (definito:alfa)

    For isk = 1 To DaTos(0).ISEAL

        If DaTos(0).seal(10, isk) = 0 Then GoTo 1986

        SEA2 = DaTos(0).seal(3, isk) / 2    'spessore

        If DaTos(0).seal(4, isk) = 0 Then DaTos(0).seal(4, isk) = DaTos(0).Gap

        Select Case DaTos(0).seal(10, isk)

        Case -6

           SecPun isk

           PXPY PX(), PY(), isk

        Case -1

           CalcDir isk

           SecPun isk

           PXPY PX(), PY(), isk

        Case -2, -3, -4, -5

           DaTos(0).seal(7, isk) = (-DaTos(0).seal(10, isk) + 2) * PI / 2

           SecPun isk

           PXPY PX(), PY(), isk

        End Select

        DisegnoSS PX(), PY(), isk, 0

1986    Next isk

End Sub
