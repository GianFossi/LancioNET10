Subroutine AggiustaTaglio

Autore:            

Data:                 22/04/2000

Scopo:

Calcola la posizione della rappresentazione in sezione del taglio diaframmi.

Chiamata da:          Sub DisTagl.

Listato:

Public Sub AggiustaTaglio(j As Integer)

        Dim ILSEGNO As Integer

        If DaTos(0).tagli(j) = 0 Then ' GoTo 1852

           ILSEGNO = 1: If j Mod 2 = 0 Then ILSEGNO = -1

           If j > 2 Then DaTos(0).tagli(j) = (DaTos(0).rr - DaTos(0).Tagl) * ILSEGNO

           If j < 3 Then DaTos(0).tagli(j) = DaTos(0).Tagl * ILSEGNO

        End If

        Select Case DaTos(0).dt(31)    'angolo taglio

          Case 0: DaTos(0).tagli(9) = 1

          Case 90: DaTos(0).tagli(9) = 2

        End Select

End Sub
