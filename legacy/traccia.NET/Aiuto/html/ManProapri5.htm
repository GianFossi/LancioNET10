Scopo:

Apre in lettura/scrittura un file temporaneo che contiene i records di tipo buc.

Chiamata da: molteplici eventi.

Listato:

Sub apri5()

      Dim LungPIP As Integer, Casuale As String

      LungPIP = Tracciatura.LungPIP

      If Len(RTrim$(Tracciatura.Nomi(LungPIP, 1))) = 0 Then

         Casuale = Monitor.Motore.Inizio.DiscoRam + "Mtrac" + Trim(Str(LungPIP))

         Tracciatura.Nomi(LungPIP, 1) = Casuale

      End If

      If Tracciatura.File5 Then Close #Tracciatura.File5

      Tracciatura.File5 = FreeFile

      Open RTrim$(Tracciatura.Nomi(LungPIP, 1)) For Random As #Tracciatura.File5 Len = 204

End Sub
