Scopo:

Verifica la condizione di non intersezione con il tubo centrale (se esistente).

Function VerifInters(p1 As Vec2, p2 As Vec2) As Integer

       Dim Vect1 As Vec2, Vect2 As Vec2

       Dim Prod1 As Single, Prod2 As Single

       Dim CentroC As RoutBase1.clsVec2

       Dim clP1 As RoutBase1.clsVec2

       Dim clP2 As RoutBase1.clsVec2

       Dim Retta As RoutBase1.clsLinea2

       Dim Punti As RoutBase1.clsPunti

       Dim iCheck As Integer, j As Integer

       Dim n1 As Integer, n2 As Integer

       If DaTos(1).cinter < DaTos(0).rtubo Then VerifInters = True: Exit Function

'verifica sormonto con tubo centrale------------------------------

       Set CentroC = New clsVec2: CentroC.x = 0: CentroC.y = 0

       Set clP1 = New RoutBase1.clsVec2: clP1.x = p1.x: clP1.y = p1.y

       Set clP2 = New RoutBase1.clsVec2: clP2.x = p2.x: clP2.y = p2.y

       Set Retta = New RoutBase1.clsLinea2

       Set Retta.P0 = clP1: Set Retta.p1 = clP2

       Retta.CalcolaDir

       Set Punti = New RoutBase1.clsPunti

       Punti.inizia 4

                   Retta.InterRettCerch DaTos(0).rtubo, CentroC, DaTos(1).cinter, Punti, n1, n2

                   iCheck = True

                   For j = 1 To 4

                   If n1 > 0 And j < 3 Or n2 > 0 And j > 2 Then

                      Vect1.x = Punti.Punti(j).x - p1.x

                      Vect1.y = Punti.Punti(j).y - p1.y

                      Vect2.x = Punti.Punti(j).x - p2.x

                      Vect2.y = Punti.Punti(j).y - p2.y

                      Prod1 = Vect1.x * Retta.Direz.x + Vect1.y * Retta.Direz.y

                      Prod2 = Vect2.x * Retta.Direz.x + Vect2.y * Retta.Direz.y

                      If Prod1 * Prod2 < 0 Then iCheck = False

                   End If

                   Next

       VerifInters = iCheck

       Set CentroC = Nothing

       Set clP1 = Nothing

       Set clP2 = Nothing

       Set Retta = Nothing

       Set Punti = Nothing

End Function
