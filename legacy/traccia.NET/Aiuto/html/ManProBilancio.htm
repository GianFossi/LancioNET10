Subroutine Bilancio

Autore:                       

Data:                        28/05/2000

Scopo:

Esegue il bilanciamento dei fori della corona esterna e della zona interna delle tracciature per fasci a fontana.

Chiama la subroutine DistInMem, la quale, tramite la compilazione di una lista ordinata di fori candidati per l’eliminazione, costituisce la parte critica dell’algoritmo di eliminazione.

Albero delle chiamate

Listato:

Sub Bilancio()

  Dim iDat As Integer, iCor As Integer

  Dim Raggio As Single, jFila As Integer

  Dim iTubo As Integer, k As Integer, j As Integer

  Dim nnn As Integer, x As Single, y As Single

  Dim iCod As Integer, Raggiol As Single

  Dim ksect As Integer, xtub As Single, ytub As Single

  Dim Obiettivo As Integer, ktot As Integer

  Dim ntubo As Integer, iT As Integer, Text As String

  RiDim

  ntubo = 0

         Text = "Riordino delle coordinate fori" + vbCrLf

  Text = Text + "in corso "

  Monitor.Motore.ProgrInizio Text

  frmTraccia.StatusBar1.Style = sbrSimple

  frmTraccia.StatusBar1.SimpleText = "Attendere, prego . . ."

  frmTraccia.StatusBar1.Refresh

  frmTraccia.Picture1.FillStyle = vbFSSolid

  DistInMem

  For iDat = 0 To 1

    Monitor.Motore.Avanzamento = (0.8 + (iDat + 1) / 2 * 0.1) * 100

    ktot = DaTos(iDat).ktotal

    Tracciatura.LungPIP = iDat + 1

    apri5

    If DaTos(0).ntubi > 0 Then

       Obiettivo = DaTos(0).ntubi

    Else

       If iDat = 0 Then Obiettivo = DaTos(1).ktotal Else Obiettivo = DaTos(0).ktotal

    End If

       For iT = 1 To ktot - Obiettivo

       GoSub Regis

       If jFila > 0 And iTubo > 0 Then

          Get #Tracciatura.File5, jFila, DaTos(iDat).bu

          Mid$(DaTos(iDat).bu.bu, iTubo, 1) = "0"

          Put #Tracciatura.File5, jFila, DaTos(iDat).bu

          frmTraccia.Picture1.Circle (xtub, ytub), DaTos(0).rtubo

          frmTraccia.Refresh

          DaTos(iDat).isett(ksect) = DaTos(iDat).isett(ksect) - 1

          DaTos(iDat).ntub(jFila) = DaTos(iDat).ntub(jFila) - 1

          DaTos(iDat).ktotal = DaTos(iDat).ktotal - 1

          If iDat = 1 And DaTos(1).cinter > 0 Then DistCentro(ntubo, iDat) = 1E+20 Else DistCentro(ntubo, iDat) = 0

       End If

       Next iT

  Next iDat

  Monitor.Motore.Avanzamento = 101

  Monitor.Motore.ProgrAmmazza

  frmTraccia.StatusBar1.Style = sbrNormal

  frmTraccia.Picture1.FillStyle = vbFSTransparent

Exit Sub

Regis:

      Raggio = Raggiol

      iTubo = Posizbu(iT, iDat)

      jFila = Fila(iT, iDat)

      ksect = Settore(iT, iDat)

      xtub = PuntiD(iT, iDat).x

      ytub = PuntiD(iT, iDat).y

      ntubo = iT

Return

End Sub
