Subroutine InitDaTos

Autore:                       

Data:                        01/05/2000

Scopo:

Inizializza la struttura di dati DaTos(0), popolandola dei dati di libreria costanti per tutta la durata delle elaborazioni.

In particolare inizializza la tabella (Tlmax) delle massime lunghezze non supportate.

Chiamata da: molteplici eventi.

Listato:

Sub InitDaTos()

   Dim SavCom As String, ifl As Integer, i As Integer

   Dim k As Integer, Riga As String

   If GiaInitDaTos Then Exit Sub

   GiaInitDaTos = True

   On Error Resume Next

   If Asc(DaTos(0).commes) < 32 Then DaTos(0).commes = " "

   SavCom = DaTos(0).commes

   Err.Clear

   On Error GoTo 0

   ReDim DaTos(0) As typDaTos

   DaTos(0).alfajp(0) = PI / 6: DaTos(0).alfajp(1) = PI / 3

   DaTos(0).alfajp(2) = PI / 2: DaTos(0).alfajp(3) = PI / 4

   DaTos(0).ips1 = 1

   DaTos(0).radius = 1!

   DaTos(0).commes = SavCom

   ifl = FreeFile

   Open RTrim$(Monitor.Motore.Inizio.Archdir) + "\TRDATA.DAT" For Input Shared As #ifl

   For i = 1 To 13

     Line Input #ifl, Riga

     DaTos(0).Dtub0(i) = Val(Riga)

   Next

   For i = 1 To 11

      Line Input #ifl, Riga

      DaTos(0).idelt(i) = Val(Riga)

   Next

   For k = 1 To 2

      For i = 1 To 13

        Line Input #ifl, Riga

        DaTos(0).Tlmax(i, k) = Val(Riga)

      Next i

   Next k

   Close #ifl: Close #Tracciatura.File5: Tracciatura.File5 = 0

   If Len(Tracciatura.Nomi(1, 1)) > 0 Then If Len(Dir$(Tracciatura.Nomi(1, 1))) > 0 Then Kill Tracciatura.Nomi(1, 1)

   If Len(Tracciatura.Nomi(2, 1)) > 0 Then If Len(Dir$(Tracciatura.Nomi(2, 1))) > 0 Then Kill Tracciatura.Nomi(2, 1)

End Sub
