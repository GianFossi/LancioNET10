Sub DiaTir()

Dim ntir As Integer, i As Integer

DaTos(0).rr = DaTos(0).di1 / 2: DaTos(0).raotl = DaTos(0).OTL / 2

ntir = 4

If DaTos(0).di1 > 381 Then ntir = 6

If DaTos(0).di1 > 838 Then ntir = 8

If DaTos(0).di1 > 1219 Then ntir = 10

If DaTos(0).ntira < ntir Then

   For i = DaTos(0).ntira + 1 To ntir

       DaTos(0).td(1, i) = 0

       DaTos(0).td(2, i) = 0

       DaTos(0).td(3, i) = 0

   Next

   DaTos(0).ntira = ntir

End If

DaTos(0).Didia = DaTos(0).di1 - 2.5                     'up to 13

If DaTos(0).di1 > 330 Then DaTos(0).Didia = DaTos(0).di1 - 3.2 'up to 17

If DaTos(0).di1 > 432 Then DaTos(0).Didia = DaTos(0).di1 - 3.8 'up to 23

If DaTos(0).di1 > 584 Then DaTos(0).Didia = DaTos(0).di1 - 4.5 'up to 39

If DaTos(0).di1 > 991 Then DaTos(0).Didia = DaTos(0).di1 - 5.7 'up to 54

If DaTos(0).di1 > 1372 Then DaTos(0).Didia = DaTos(0).di1 - 7.5 'up to 60

If DaTos(0).di1 > 1524 Then DaTos(0).Didia = DaTos(0).di1 - 9

End Sub
