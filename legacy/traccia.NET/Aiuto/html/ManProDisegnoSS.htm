Subroutine DisegnoSS

Autore:            

Data:                 22/04/2000

Scopo:

Disegna la sezione retta di una sealing strip.

Chiamata da:          Sub Sealing, Sub MettiSS, Form frmTraccia.

Listato:

Public Sub DisegnoSS(PX() As Single, PY() As Single, isk As Integer, Mode As Integer)

     Dim Rx1 As Single, Rx2 As Single, Ry1 As Single, Ry2 As Single

     Dim n As Integer, Tag As String

     Dim twLeft As Single, twTop As Single, twWidth As Single, twHeight As Single

     'Mode 0 da SubTraccia,1 da SubVista (non esiste più)

     Rx1 = Min4(PX())

     Rx2 = Max4(PX())

     Ry1 = Min4(PY())

     Ry2 = Max4(PY())

     If VarType(frmTraccia.Pix(isk)) = vbObject Then ' Is Nothing Then

        Load frmTraccia.Pix(isk)

        frmTraccia.Pix(isk).ScaleMode = vbPixels

     Else

        If Mode = 1 Then

             With frmTraccia.Pix(isk)

                n = InStr(.Tag, "|")

                Tag = .Tag

                twTop = Val(Left$(Tag, n - 1))

                Tag = Right$(Tag, Len(Tag) - n)

                n = InStr(Tag, "|")

                twLeft = Val(Left$(Tag, n - 1))

                Tag = Right$(Tag, Len(Tag) - n)

                n = InStr(Tag, "|")

                twHeight = Val(Left$(Tag, n - 1))

                twWidth = Val(Right$(Tag, Len(Tag) - n))

                frmTraccia.Picture1.ScaleMode = vbPixels 'vbTwips

                frmTraccia.Picture1.PaintPicture .Image, twLeft, twTop, twWidth, twHeight, 0, 0, twWidth, twHeight

             End With

             Monitor.Routines.Scala -DaTos(0).XSC1, DaTos(0).XSC1, -DaTos(0).Ysc1, DaTos(0).Ysc1, , True

        End If

     End If

     twHeight = (Ry2 - Ry1) / frmTraccia.Picture1.ScaleHeight * frmTraccia.pictHeightpx

     twWidth = (Rx2 - Rx1) / frmTraccia.Picture1.ScaleWidth * frmTraccia.pictWidthpx

     twTop = (Ry1 - frmTraccia.Picture1.ScaleTop) / frmTraccia.Picture1.ScaleHeight * frmTraccia.pictHeightpx

     twLeft = (Rx1 - frmTraccia.Picture1.ScaleLeft) / frmTraccia.Picture1.ScaleWidth * frmTraccia.pictWidthpx

     If twHeight < 0 Then twHeight = -twHeight: twTop = twTop - twHeight

     If twWidth < 0 Then twWidth = -twWidth: twLeft = twLeft - twWidth

     twTop = CInt(twTop) - 4: twLeft = CInt(twLeft) - 4

     twHeight = CInt(twHeight) + 8: twWidth = CInt(twWidth) + 8

     frmTraccia.Picture1.ScaleMode = vbPixels ' vbTwips

     If twTop < 0 Then twTop = 0

     If twLeft < 0 Then twLeft = 0

     If twTop + twHeight > frmTraccia.Picture1.ScaleHeight Then twHeight = frmTraccia.Picture1.ScaleHeight - twTop

     If twLeft + twWidth > frmTraccia.Picture1.ScaleWidth Then twWidth = frmTraccia.Picture1.ScaleWidth - twWidth

     With frmTraccia.Pix(isk)

        .Height = twHeight + 50 '.ScaleHeight .scalemode

        .Width = twWidth + 50

        .Cls

        .Tag = Str$(twTop) + "|" + Str$(twLeft) + "|" + Str$(twHeight) + "|" + Str$(twWidth)

        .AutoRedraw = True

       ' .Visible = True

       ' .ZOrder

        .PaintPicture frmTraccia.Picture1.Image, 0, 0, twWidth, twHeight, twLeft, twTop, twWidth, twHeight

     End With

     Monitor.Routines.Scala -DaTos(0).XSC1, DaTos(0).XSC1, -DaTos(0).Ysc1, DaTos(0).Ysc1, , True

     If Mode = 0 Then

        Monitor.Routines.tratto PX(1), PY(1), PX(2), PY(2), 0.3, 0

        Monitor.Routines.tratto PX(2), PY(2), PX(3), PY(3), 0.3, 0

        Monitor.Routines.tratto PX(3), PY(3), PX(4), PY(4), 0.3, 0

        Monitor.Routines.tratto PX(4), PY(4), PX(1), PY(1), 0.3, 0

     Else

        frmTraccia.Picture1.DrawStyle = vbSolid

        frmTraccia.Picture1.Line (PX(1), PY(1))-(PX(2), PY(2)), QBColor(4)

        frmTraccia.Picture1.Line -(PX(3), PY(3)), QBColor(4)

        frmTraccia.Picture1.Line -(PX(4), PY(4)), QBColor(4)

        frmTraccia.Picture1.Line -(PX(1), PY(1)), QBColor(4)

     End If

End Sub
