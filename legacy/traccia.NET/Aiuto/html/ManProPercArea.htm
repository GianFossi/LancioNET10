Subroutine PercArea

Autore:            

Data:                 22/04/2000

Scopo:

Calcola la quota di taglio dei diaframmi quando è nota la percentuale di taglio espressa come superficie.

Chiamata da:          Sub inpiDIA.

Listato:

Sub PercArea(toltu As Single)

      Dim atu As Single, VPA As Single

      Dim alor1 As Single, anet1 As Single

      Dim ktotu As Integer, ralibe As Single

      Dim atfre As Single, perfre As Single

      Dim difs As Single, ntui As Integer

      Dim akh As Single, rbt As Single

      Dim bet As Single, alord As Single, alibe As Single

      Dim ntt As Integer, ICER As Integer

      Dim H As Single, inh As Integer, ifh As Integer

      Dim jk As Integer, difs0 As Single

      Dim tagl0 As Single, ak As Single

      Dim antui As Single, j As Integer

      Dim alord1 As Single, OPA As Single

      Dim yj As Single, hvi As Single

      toltu = DaTos(0).dtubo * 0.3: VPA = DaTos(0).VPASSO / 2

      atu = DaTos(0).dtubo ^ 2 * PI / 4

      alor1 = DaTos(0).OPASSO * DaTos(0).VPASSO

      anet1 = alor1 - atu: ralibe = anet1 / alor1

210   If DaTos(0).dt(31) = 0 Then GoSub tglareah

      If DaTos(0).dt(31) = 90 Then GoSub tglareav

      ktotu = DaTos(0).ktotal

      If DaTos(0).JSHELL = 3 Then ktotu = DaTos(0).ktotal * 2

      atfre = PI / 4 * (DaTos(0).di1 ^ 2 - ktotu * DaTos(0).dtubo ^ 2)

      perfre = atfre * Abs(DaTos(0).dt(30)) / 100

      difs = 0: ntui = 0: akh = 0

220:  rbt = DaTos(0).Tagl / DaTos(0).rr

      bet = acos(rbt)

      alord = bet * DaTos(0).rr ^ 2 - Sin(bet) * DaTos(0).rr * DaTos(0).Tagl

      If DaTos(0).dt(31) = 90 Then GoTo verti

      ntui = 0: ntt = 0: ICER = 1

      If akh = 0 Then ntt = 0.5 * (DaTos(0).ntub(H) + DaTos(0).ntus(H))

      inh = H: ifh = DaTos(0).kymax

      If DaTos(0).iutu = 1 Then inh = 1: ifh = H: ICER = 0

      For jk = inh To ifh: ntui = ntui + DaTos(0).ntub(jk) + DaTos(0).ntus(jk) * ICER: Next jk

      ntui = ntui - ntt

230:  alibe = alord - ntui * atu

      If alibe <> perfre Then

         difs0 = perfre - alibe

         If Sgn(difs0) + Sgn(difs) = 0 Then GoTo 4578

         tagl0 = DaTos(0).Tagl: difs = difs0

         ak = 0.5: If DaTos(0).iutu = 1 Then ak = -0.5

         akh = akh - ak * Sgn(difs): H = Fix(H + akh + 0.5)

         If Abs(akh) >= 1 Then akh = 0

         DaTos(0).Tagl = DaTos(0).Tagl - VPA * Sgn(difs)

         GoTo 220

      End If

4578  If Abs(difs0) > Abs(difs) Then DaTos(0).Tagl = tagl0

      If DaTos(0).Tagl < 0 Then DaTos(0).Tagl = -DaTos(0).Tagl

      Exit Sub

verti: antui = 0

      For j = 1 To DaTos(0).kymax

         If DaTos(0).ntub(j) = 0 Or DaTos(0).Tagl > DaTos(0).xf(j) Then GoTo 4021

         antui = antui + 1 + (DaTos(0).xf(j) - DaTos(0).Tagl) / DaTos(0).OPASSO

4021  Next j

      If DaTos(0).iutu = 1 Then antui = 2 * antui

      alord1 = atu * antui

      alibe = alord - alord1

      If alibe <> perfre Then

         difs0 = perfre - alibe

         If Sgn(difs0) + Sgn(difs) = 0 Then GoTo 4577

         tagl0 = DaTos(0).Tagl: difs = difs0

         DaTos(0).Tagl = DaTos(0).Tagl - OPA * Sgn(difs)

         GoTo 220

      End If

4577  If Abs(difs0) > Abs(difs) Then DaTos(0).Tagl = tagl0

      Exit Sub

tglareah:

      For H = DaTos(0).kymax To 1 Step -1

          yj = Abs(DaTos(0).Y(H))

          If yj < DaTos(0).Tagl + DaTos(0).dtubo And yj > DaTos(0).Tagl - DaTos(0).dtubo Then

             DaTos(0).Tagl = yj: Return

          End If

      Next H

      BEEP

      Return

tglareav:

      OPA = DaTos(0).OPASSO / 2 ': IF DaTos(0).JP = 4 THEN opa = DaTos(0).OPASSO

      For hvi = DaTos(0).xmind To 1 Step OPA

          If Abs(hvi) < DaTos(0).Tagl + toltu Or DaTos(0).Tagl - toltu > Abs(hvi) Then DaTos(0).Tagl = Abs(hvi): Return

      Next hvi

      BEEP

      Return

End Sub
