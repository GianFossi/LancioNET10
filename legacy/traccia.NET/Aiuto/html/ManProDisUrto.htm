Subroutine DisUrto

Autore:            

Data:                 22/04/2000

Scopo:

Disegna la sezione dell’eventuale piatto d’urto nella pianta della tracciatura.

Chiamata da:          Sub inpiDIA.

Listato:

Sub DisUrto()

      Dim YURT0 As Single, YURT1 As Single

      Dim XURT As Single, XCON As Single

      If DaTos(0).dt(33) = 0 Or DaTos(0).idout = 207 And DaTos(0).URTY(1) = 0 Then Exit Sub

      If DaTos(0).URTY(1) = 0 Then

         If DaTos(0).iutu = 1 Then YURT0 = (-DaTos(0).Y(1) + DaTos(0).rtubo + 3): GoTo ava

         YURT0 = (DaTos(0).Y(DaTos(0).kymax) + DaTos(0).rtubo + 3)

      Else

         YURT0 = DaTos(0).URTY(1)

      End If

ava:

'-------------------------------------------------

      If YURT0 > DaTos(0).rr Then

         If YURT0 = DaTos(0).URTY(1) Then

            YURT0 = 0.8 * DaTos(0).rr

            DaTos(0).URTY(1) = YURT0

         Else

            YURT0 = 0.8 * DaTos(0).rr

         End If

      End If

'------------------------------------------------

      If DaTos(0).URTY(3) = 0 Then DaTos(0).URTY(3) = 10

      YURT1 = YURT0 + DaTos(0).URTY(3)

      If DaTos(0).URTY(2) <> 0 Then

         XURT = DaTos(0).URTY(2) / 2

      Else

         XURT = DaTos(0).hin * Tan(PI / 6) + DaTos(0).dt(3) / 2

         If XURT = 0 Then XURT = 100

         DaTos(0).URTY(2) = XURT

      End If

      XCON = Sqr(DaTos(0).rr * DaTos(0).rr - YURT0 * YURT0)

      If XURT + 25 > XCON Then XURT = XCON - 25

      Monitor.Routines.trattoBF -XURT, YURT1, XURT, YURT0, 0.2, 0

      DaTos(0).URTY(1) = YURT0

      DaTos(0).URTY(2) = 2 * XURT

      DaTos(0).URTY(4) = DaTos(0).URTY(2)

End Sub
