Sub inpiDIA()

Dim toltu As Single

200   If DaTos(0).dt(30) <> 0 And DaTos(0).dt(32) < 5 Then

         If DaTos(0).dt(32) <> 1 Then DaTos(0).Tagl = DaTos(0).rr * (1 - Abs(DaTos(0).dt(30)) / 50)

         If DaTos(0).dt(30) < 0 Then Call PercArea(toltu)   'percentuale su area

1876     Select Case DaTos(0).dt(32)

            Case 0, 1, 5: Call DisTagl(2)      'single/ NTW /support

            Case 2:    Call DisTagl(4)       'Double segmental

            Case Else: ' PRINT "Tipo diaframmi"; DaTos(0).dt(32); "Da fare": u$ = INPUT$(1)

         End Select

         Call DisTaglSupp

      End If

      Call DisUrto

End Sub
